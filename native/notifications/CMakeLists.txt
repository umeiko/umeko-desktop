cmake_minimum_required(VERSION 3.24)
project(UmekoNotifications LANGUAGES CXX)
option(UMEKO_NOTIFY_SHARED "Build a library for the Python host" ON)
set(kind STATIC)
if(UMEKO_NOTIFY_SHARED)
  set(kind SHARED)
endif()
add_library(umeko_notifications ${kind} events.cpp)
target_compile_features(umeko_notifications PUBLIC cxx_std_17)
target_include_directories(umeko_notifications PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(WIN32 AND MSVC)
  target_sources(umeko_notifications PRIVATE notify_windows.cpp wintoast/wintoastlib.cpp)
  target_include_directories(umeko_notifications PRIVATE wintoast)
  target_link_libraries(umeko_notifications PRIVATE psapi shlwapi user32 ole32 propsys shell32)
  target_compile_definitions(umeko_notifications PRIVATE NOMINMAX UNICODE _UNICODE)
  target_compile_options(umeko_notifications PRIVATE /utf-8)
  if(UMEKO_NOTIFY_SHARED)
    set_property(TARGET umeko_notifications PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
elseif(APPLE)
  enable_language(OBJCXX)
  set_target_properties(umeko_notifications PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED ON)
  target_sources(umeko_notifications PRIVATE notify_macos.mm)
  target_compile_options(umeko_notifications PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
  target_link_libraries(umeko_notifications PRIVATE "-framework Foundation" "-framework UserNotifications")
else()
  target_sources(umeko_notifications PRIVATE notify_stub.cpp)
endif()
